%%  Inline code.  \ct is short enough to use constantly, which is the point.
%%  It is \lstinline, so ^ # & % _ $ inside it need no escaping -- which
%%  matters in a book about a language whose return statement is a caret.
%%  It cannot appear in a section title or any other moving argument; \ctp
%%  is the plain-text fallback for those, and there you escape by hand.
\lstdefinestyle{stinline}{
  language=Smalltalk2026,
  basicstyle=\ttfamily\small\color{stink},
  keywordstyle=\color{stink},
  commentstyle=\color{stgrey}\itshape,
  stringstyle=\color{stink},
  showstringspaces=false,
  columns=fullflexible,
  keepspaces=true,
  breaklines=false,
}
\newcommand{\ct}{\lstinline[style=stinline]}
\newcommand{\ctp}[1]{\texorpdfstring{\mbox{\ttfamily\small #1}}{#1}}
